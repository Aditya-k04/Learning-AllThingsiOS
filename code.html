    Custom View

    🎯 Quest: Extract a reusable AvatarBadge that turns a name into its initials inside a colored circle, then use it twice.

    Show the code 
    import SwiftUI

struct AvatarBadge: View {
    let name: String

    private var initials: String {
        name.split(separator: " ")
            .prefix(2)
            .compactMap { $0.first }
            .map(String.init)
            .joined()
            .uppercased()
    }

    var body: some View {
        Text(initials)
            .font(.headline)
            .foregroundStyle(.white)
            .frame(width: 44, height: 44)
            .background(.teal, in: .circle)
    }
}

#Preview {
    HStack {
        AvatarBadge(name: "Ada Lovelace")
        AvatarBadge(name: "Grace Hopper")
    }
}

    A view is just a struct conforming to View. Pulling repeated UI into its own type makes it reusable and testable.

    Programmatic Navigation

    🎯 Quest: Drive a NavigationStack with a @State path so you can push and pop in code.

    Show the code 
    import SwiftUI

enum Route: Hashable {
    case detail(Int)
    case settings
}

struct ProgrammaticNav: View {
    @State private var path: [Route] = []

    var body: some View {
        NavigationStack(path: $path) {
            VStack(spacing: 16) {
                Button("Go to detail 42") { path.append(.detail(42)) }
                Button("Go to settings") { path.append(.settings) }
            }
            .navigationDestination(for: Route.self) { route in
                switch route {
                case .detail(let id):
                    Button("Pop to root") { path.removeAll() }
                        .navigationTitle("Detail \(id)")
                case .settings:
                    Text("Settings")
                }
            }
        }
    }
}

#Preview {
    ProgrammaticNav()

    Binding the stack to path makes navigation state data you own — appending pushes, removing pops, and removeAll() jumps back to the root.

    Async Load with a Loading State

    🎯 Quest: Load data in .task { } and model the phases with a .loading / .loaded / .failed enum.

    Show the code 
    import SwiftUI

struct AsyncDemo: View {
    enum Phase {
        case loading
        case loaded(String)
        case failed(String)
    }

    @State private var phase: Phase = .loading

    var body: some View {
        VStack {
            switch phase {
            case .loading:
                ProgressView("Loading…")
            case .loaded(let text):
                Text(text).font(.title2)
            case .failed(let message):
                Text("Error: \(message)").foregroundStyle(.red)
            }
        }
        .task {
            do {
                phase = .loaded(try await fetchGreeting())
            } catch {
                phase = .failed(error.localizedDescription)
            }
        }
    }

    func fetchGreeting() async throws -> String {
        try await Task.sleep(for: .seconds(1))
        return "Hello from async!"
    }
}

#Preview {
    AsyncDemo()
}

    .task runs an async job tied to the view's lifetime — it auto-cancels when the view disappears. A phase enum makes every state (including failure) explicit and unrepresentable-when-invalid.

    Animation

    🎯 Quest: Show three animation styles — withAnimation, implicit .animation(_:value:), and a .transition on insert/remove.

    Show the code 
    import SwiftUI

struct AnimationDemo: View {
    @State private var expanded = false
    @State private var showBox = false

    var body: some View {
        VStack(spacing: 24) {
            // Explicit
            Button("Toggle size") {
                withAnimation(.spring) { expanded.toggle() }
            }
            Circle()
                .fill(.orange)
                .frame(width: expanded ? 120 : 60, height: expanded ? 120 : 60)

            // Implicit
            Rectangle()
                .fill(expanded ? .green : .blue)
                .frame(width: 120, height: 20)
                .animation(.easeInOut, value: expanded)

            // Transition
            Button("Toggle box") { showBox.toggle() }
            if showBox {
                Text("Hello")
                    .padding()
                    .background(.yellow, in: .rect(cornerRadius: 8))

            }
        }
        .animation(.default, value: showBox)
    }
}

#Preview {
    AnimationDemo()
}

    withAnimation animates the state change at the call site; .animation(_:value:) animates whenever that value changes; .transition describes how a view enters and leaves when it's inserted or removed.

    Create and lay out a view in code

    🎯 Quest: In loadView/viewDidLoad, add a centered label above a button and pin them with Auto Layout anchors.

    Show the code 
    final class RootViewController: UIViewController {
    override func viewDidLoad() {
        super.viewDidLoad()
        view.backgroundColor = .systemBackground

        let label = UILabel()
        label.text = "Hello, UIKit"
        label.font = .preferredFont(forTextStyle: .largeTitle)
        label.translatesAutoresizingMaskIntoConstraints = false

            label.centerXAnchor.constraint(equalTo: view.centerXAnchor),
            label.centerYAnchor.constraint(equalTo: view.centerYAnchor),

    Setting translatesAutoresizingMaskIntoConstraints = false opts each view out of the legacy autoresizing bridge so your anchor constraints are the single source of truth; NSLayoutConstraint.activate batches them efficiently.

    Subclass a reusable UIView

    🎯 Quest: Build an AvatarView that works from code, supports both initializers via a shared commonInit, and exposes configure(name:).

    Show the code 
    final class AvatarView: UIView {
    private let initialsLabel = UILabel()

    override init(frame: CGRect) {
        super.init(frame: frame)
        commonInit()
    }

    required init?(coder: NSCoder) {
        super.init(coder: coder)
        commonInit()
    }

    private func commonInit() {
        backgroundColor = .systemTeal
        layer.cornerRadius = 24
        clipsToBounds = true

        initialsLabel.textColor = .white
        initialsLabel.font = .boldSystemFont(ofSize: 18)
        initialsLabel.translatesAutoresizingMaskIntoConstraints = false
        addSubview(initialsLabel)
        NSLayoutConstraint.activate([
            initialsLabel.centerXAnchor.constraint(equalTo: centerXAnchor),
            initialsLabel.centerYAnchor.constraint(equalTo: centerYAnchor),
        ])
    }

    func configure(name: String) {
        initialsLabel.text = name.split(separator: " ")
            .compactMap { $0.first }.prefix(2)

    Both init(frame:) and the required init?(coder:) funnel into commonInit, so the view is identical whether created in code or decoded from a nib — the standard way to avoid duplicated setup.

final class PeopleViewController: UITableViewController {

    override func viewDidLoad() {
        super.viewDidLoad()
        tableView.register(PersonCell.self, forCellReuseIdentifier: PersonCell.reuseID)

        people.count

    override func tableView(_ tableView: UITableView,

    Passing data back — delegate and closure

    🎯 Quest: Send a value from a child controller back to its parent two ways: a weak delegate protocol and a completion closure.

    Show the code 
    // 1. Delegate approach
protocol EditorViewControllerDelegate: AnyObject {
    func editor(_ editor: EditorViewController, didSave text: String)
}

final class EditorViewController: UIViewController {
    weak var delegate: EditorViewControllerDelegate?

    // 2. Closure approach — same job, less ceremony
    var onSave: ((String) -> Void)?

    private func save(_ text: String) {
        delegate?.editor(self, didSave: text)   // delegate path
        onSave?(text)                           // closure path
    }
}

    The delegate is weak (and the protocol is AnyObject) to break the retain cycle between parent and child. Closures are lighter for one-off callbacks; delegates scale better when the child reports many distinct events.

    Gestures

    🎯 Quest: Let the user drag a view around with a pan, and reset it to center on a tap.

    Show the code 
    let pan = UIPanGestureRecognizer(target: self, action: #selector(handlePan))
let tap = UITapGestureRecognizer(target: self, action: #selector(handleTap))
card.addGestureRecognizer(pan)
card.addGestureRecognizer(tap)

@objc private func handlePan(_ gesture: UIPanGestureRecognizer) {
    guard let view = gesture.view else { return }
    let translation = gesture.translation(in: view.superview)
    switch gesture.state {
    case .changed:
        view.center = CGPoint(x: view.center.x + translation.x,
                              y: view.center.y + translation.y)
        gesture.setTranslation(.zero, in: view.superview)
    case .ended, .cancelled:
        break
    default:
        break
    }
}

@objc private func handleTap(_ gesture: UITapGestureRecognizer) {
    guard let view = gesture.view, let bounds = view.superview?.bounds else { return }
    UIView.animate(withDuration: 0.3) {
        view.center = CGPoint(x: bounds.midX, y: bounds.midY)
    }
}

    Resetting translation to .zero on every .changed makes each callback report only the incremental delta, so you add it to the current center instead of re-applying the total drag each frame.

    🎯 Quest: Broadcast that data updated via NotificationCenter, and observe it elsewhere.

    Show the code 
    extension Notification.Name {

}

    self?.refresh()
}

    NotificationCenter is best for one-to-many, loosely coupled events. For watching a single object's property, reach for KVO instead — observe(\.property) { object, change in ... } gives you type-safe old/new values and auto-invalidates when the returned token is released.

    Bridge to SwiftUI

    🎯 Quest: Embed a SwiftUI view inside a UIKit controller using UIHostingController and view-controller containment.

    Show the code 
    import SwiftUI

func embedSwiftUI() {
    let hosting = UIHostingController(rootView: ProfileView())
    addChild(hosting)
    hosting.view.translatesAutoresizingMaskIntoConstraints = false
    view.addSubview(hosting.view)
    NSLayoutConstraint.activate([
        hosting.view.topAnchor.constraint(equalTo: view.safeAreaLayoutGuide.topAnchor),
        hosting.view.leadingAnchor.constraint(equalTo: view.leadingAnchor),
        hosting.view.trailingAnchor.constraint(equalTo: view.trailingAnchor),
        hosting.view.bottomAnchor.constraint(equalTo: view.bottomAnchor),
    ])
    hosting.didMove(toParent: self)
}

    Follow the full containment dance — addChild, add the view + constraints, then didMove(toParent:) — so the hosting controller receives lifecycle and trait events correctly. Skipping didMove leaves the child in a half-attached state.

    Write it before you peek. These quests all use modern stitchable SwiftUI shaders — [[ stitchable ]] Metal functions loaded through ShaderLibrary and applied with .colorEffect, .distortionEffect, and .layerEffect.

    Work top to bottom. These quests are a ladder — each one assumes the ones above it, so you build skills and see how the pieces fit together.

    A distortion ripple

    🎯 Quest: Use .distortionEffect to push pixels outward in a ring that expands from a tapped origin over time. Remember maxSampleOffset.

    Show the code 
    [[ stitchable ]]
float2 ripple(float2 position, float2 origin, float time) {
    float d = distance(position, origin);
    float wave = sin(d * 0.1 - time * 6.0) * exp(-d * 0.01) * 12.0;
    float2 dir = normalize(position - origin);
    return position + dir * wave;
}

    Image("photo")
    .distortionEffect(
        ShaderLibrary.ripple(
            .float2(tapOrigin),
            .float(time)
        ),
        maxSampleOffset: CGSize(width: 12, height: 12)
    )

    A distortionEffect returns the source position to sample for each output pixel, so you return position + displacement. maxSampleOffset tells SwiftUI the largest distance you'll reach, so it can allocate enough surrounding pixels — undersize it and the ripple gets clipped.

    A layer effect blur/glow

    🎯 Quest: Use .layerEffect to sample neighboring pixels via SwiftUI::Layer.sample and average them into a soft blur/glow.

    Show the code 
    [[ stitchable ]]
half4 boxBlur(float2 position, SwiftUI::Layer layer) {
    half4 sum = half4(0.0h);
    for (float y = -2.0; y <= 2.0; y += 1.0) {
        for (float x = -2.0; x <= 2.0; x += 1.0) {
            sum += layer.sample(position + float2(x, y) * 2.0);
        }
    }
    return sum / 25.0;
}

    Image("photo")
    .layerEffect(
        ShaderLibrary.boxBlur(),
        maxSampleOffset: CGSize(width: 4, height: 4)
    )

    A layerEffect hands you the whole rendered layer, so you can read other pixels — impossible in a colorEffect. Averaging a grid of layer.sample calls around each pixel is the simplest blur; boosting the sum instead of dividing gives a glow.

    Value noise

    🎯 Quest: Build a hash-based value noise function and map its output to grayscale.

    Show the code 
    float hash(float2 p) {
    p = fract(p * float2(123.34, 456.21));
    p += dot(p, p + 45.32);
    return fract(p.x * p.y);
}

float valueNoise(float2 p) {
    float2 i = floor(p);
    float2 f = fract(p);
    float2 u = f * f * (3.0 - 2.0 * f);
    float a = hash(i);
    float b = hash(i + float2(1.0, 0.0));
    float c = hash(i + float2(0.0, 1.0));
    float d = hash(i + float2(1.0, 1.0));
    return mix(mix(a, b, u.x), mix(c, d, u.x), u.y);
}

[[ stitchable ]]
half4 noiseView(float2 position, half4 color, float2 size) {
    float2 uv = position / size;
    float n = valueNoise(uv * 8.0);
    return half4(half(n));
}

    Rectangle()
    .frame(width: 256, height: 256)
    .visualEffect { content, proxy in
        content.colorEffect(
            ShaderLibrary.noiseView(.float2(proxy.size))
        )
    }

    hash turns a grid corner into a pseudo-random value; valueNoise interpolates between the four surrounding corners with a smooth (3f² - 2f³) curve so the result has no blocky seams. Scaling uv sets the noise frequency.

    fBm

    🎯 Quest: Sum several octaves of the value noise (fractional Brownian motion) to make soft cloud texture.

    Show the code 
    float fbm(float2 p) {
    float total = 0.0;
    float amp = 0.5;
    for (int i = 0; i < 5; i++) {
        total += valueNoise(p) * amp;
        p *= 2.0;
        amp *= 0.5;
    }
    return total;
}

[[ stitchable ]]
half4 clouds(float2 position, half4 color, float2 size) {
    float2 uv = position / size;
    float f = fbm(uv * 4.0);
    return half4(half(f), half(f), half(f * 1.1), 1.0h);
}

    Rectangle()
    .frame(width: 256, height: 256)
    .visualEffect { content, proxy in
        content.colorEffect(
            ShaderLibrary.clouds(.float2(proxy.size))
        )
    }

    fBm layers the same noise at doubling frequency and halving amplitude. Big octaves give the broad shapes, small octaves add fine detail — the sum reads like clouds or smoke.

    Voronoi sparks

    🎯 Quest: Split the plane into cells, find the distance to the nearest random cell point (F1), and light up bright dots.

    Show the code 
    float2 hash2(float2 p) {
    p = float2(dot(p, float2(127.1, 311.7)),
               dot(p, float2(269.5, 183.3)));
    return fract(sin(p) * 43758.5453);
}

[[ stitchable ]]
half4 voronoi(float2 position, half4 color, float2 size) {
    float2 uv = position / size * 8.0;
    float2 cell = floor(uv);
    float f1 = 1.0;
    for (float y = -1.0; y <= 1.0; y += 1.0) {
        for (float x = -1.0; x <= 1.0; x += 1.0) {
            float2 neighbor = float2(x, y);
            float2 point = hash2(cell + neighbor);
            float d = distance(uv, cell + neighbor + point);
            f1 = min(f1, d);
        }
    }
    float spark = 1.0 - smoothstep(0.0, 0.08, f1);
    return half4(half(spark));
}

    Rectangle()
    .frame(width: 256, height: 256)
    .visualEffect { content, proxy in
        content.colorEffect(
            ShaderLibrary.voronoi(.float2(proxy.size))
        )
    }

    Each cell owns one random feature point; F1 is the distance to the closest point across the 3×3 neighborhood. Small F1 means you're near a point, so smoothstep near zero paints a bright spark there.

    Domain warping

    🎯 Quest: Feed fBm back into itself — fbm(uv + fbm(uv)) — for a flowing, marbled look.

    Show the code 
    [[ stitchable ]]
half4 domainWarp(float2 position, half4 color, float2 size, float time) {
    float2 uv = position / size * 4.0;
    float2 q = float2(fbm(uv + float2(0.0, time * 0.1)),
                      fbm(uv + float2(5.2, 1.3)));
    float f = fbm(uv + q);
    half3 col = mix(half3(0.1h, 0.2h, 0.5h),
                    half3(0.9h, 0.7h, 0.3h), half(f));
    return half4(col, 1.0h);
}

    TimelineView(.animation) { timeline in
    let t = timeline.date.timeIntervalSinceReferenceDate
    Rectangle()
        .frame(width: 256, height: 256)
        .visualEffect { content, proxy in
            content.colorEffect(

                    .float2(proxy.size),
                    .float(t)

        }
}

    Domain warping distorts the input coordinates with noise before sampling more noise. That feedback bends the texture into swirling, organic flows; nudging the warp offset with time makes it drift.

    Pass a touch point in

    🎯 Quest: Draw a glow that follows the user's finger by passing a .float2 tap location from SwiftUI @State.

    Show the code 
    [[ stitchable ]]
half4 touchGlow(float2 position, half4 color, float2 touch) {

    float glow = exp(-d * 0.01);
    half4 tint = half4(0.3h, 0.8h, 1.0h, 1.0h) * half(glow);

    struct GlowView: View {
